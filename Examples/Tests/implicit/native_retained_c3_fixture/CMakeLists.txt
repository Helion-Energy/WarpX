cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXNativeRetainedC3Fixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NATIVE_RETAINED_C3_EXECUTABLE)
    add_executable(test_native_retained_c3 IMPORTED GLOBAL)
    set_target_properties(test_native_retained_c3 PROPERTIES IMPORTED_LOCATION "${NATIVE_RETAINED_C3_EXECUTABLE}")
elseif(TARGET lib_rz)
    add_executable(test_native_retained_c3 main.cpp)
    target_link_libraries(test_native_retained_c3 PRIVATE lib_rz)
    target_include_directories(test_native_retained_c3 PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(WarpX_COMPUTE STREQUAL "CUDA")
        setup_target_for_cuda_compilation(test_native_retained_c3)
    endif()
else()
    message(FATAL_ERROR "Requires lib_rz or NATIVE_RETAINED_C3_EXECUTABLE")
endif()
# GPU keeps exact in-process restoration; independent run comparisons have a
# separately qualified arithmetic contract and are not registered here.
if(WarpX_COMPUTE STREQUAL "CUDA" OR NOT NATIVE_RETAINED_C3_PLUGIN OR NOT NATIVE_RETAINED_C3_CONFIG)
    return()
endif()
function(c3_case name)
    add_test(NAME native_retained_c3_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run.py --executable $<TARGET_FILE:test_native_retained_c3>
        --plugin ${NATIVE_RETAINED_C3_PLUGIN} --config ${NATIVE_RETAINED_C3_CONFIG}
        --case ${name} ${ARGN})
    set_tests_properties(native_retained_c3_${name} PROPERTIES TIMEOUT 600 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
c3_case(coupled_mm0_first --fault before)
c3_case(decoupled_mm0_warm --mode decoupled_jfnk --reject-step 1)
c3_case(coupled_mm1_density --mm 1 --fault density)
c3_case(decoupled_mm1_nonfinite_warm --mode decoupled_jfnk --mm 1 --fault nonfinite --reject-step 1)
c3_case(guard_combined_vacuum --negative combined_vacuum)
if(NATIVE_RETAINED_C3_OLD_PLUGIN)
    c3_case(guard_old_provider --negative provider --old-plugin ${NATIVE_RETAINED_C3_OLD_PLUGIN})
endif()
if(MPIEXEC_EXECUTABLE)
    c3_case(mpi_coupled_mm0_relocate --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --relocate)
    c3_case(mpi_decoupled_mm0_empty --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mode decoupled_jfnk --veto-rank 1 --reject-step 1 --empty-rank)
    c3_case(mpi_coupled_mm1_density --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mm 1 --fault density)
    c3_case(mpi_decoupled_mm1_relocate_warm --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mode decoupled_jfnk --mm 1 --relocate --reject-step 1)
    set_tests_properties(native_retained_c3_mpi_coupled_mm0_relocate native_retained_c3_mpi_decoupled_mm0_empty native_retained_c3_mpi_coupled_mm1_density native_retained_c3_mpi_decoupled_mm1_relocate_warm PROPERTIES PROCESSORS 2)
endif()
