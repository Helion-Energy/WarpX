cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXNativePECPlasmaFixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NATIVE_PEC_PLASMA_EXECUTABLE)
    add_executable(test_native_pec_plasma IMPORTED GLOBAL)
    set_target_properties(test_native_pec_plasma PROPERTIES IMPORTED_LOCATION "${NATIVE_PEC_PLASMA_EXECUTABLE}")
elseif(TARGET lib_rz)
    add_executable(test_native_pec_plasma main.cpp)
    target_link_libraries(test_native_pec_plasma PRIVATE lib_rz)
    target_include_directories(test_native_pec_plasma PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(WarpX_COMPUTE STREQUAL "CUDA")
        setup_target_for_cuda_compilation(test_native_pec_plasma)
    endif()
else()
    message(FATAL_ERROR "Requires lib_rz or NATIVE_PEC_PLASMA_EXECUTABLE")
endif()
# Exact event recomputation is a CPU certificate. CUDA compile/link is covered;
# native atomic-scatter repeat equality is not assumed. GPU runtime requires its
# own independent arithmetic certificate before tests are registered here.
if(WarpX_COMPUTE STREQUAL "CUDA")
    return()
endif()
function(pec_case name)
    add_test(NAME native_pec_plasma_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run_native.py --executable $<TARGET_FILE:test_native_pec_plasma>
        --input ${CMAKE_CURRENT_SOURCE_DIR}/inputs_pec --case ${name} ${ARGN})
    set_tests_properties(native_pec_plasma_${name} PROPERTIES TIMEOUT 240 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
foreach(mode IN ITEMS coupled_jfnk decoupled_jfnk)
    pec_case(${mode}_periodic --mode ${mode})
    pec_case(${mode}_pmc --mode ${mode} --boundary pmc)
endforeach()
foreach(fault IN ITEMS alias width eta mm)
    pec_case(reject_${fault} --negative ${fault})
endforeach()
if(MPIEXEC_EXECUTABLE)
    pec_case(mpi_empty --mode decoupled_jfnk --boundary pmc --ranks 2 --mpi ${MPIEXEC_EXECUTABLE})
    pec_case(mpi_seam --mode decoupled_jfnk --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --grid 16)
    set_tests_properties(native_pec_plasma_mpi_empty native_pec_plasma_mpi_seam PROPERTIES PROCESSORS 2)
endif()
