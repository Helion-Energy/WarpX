cmake_minimum_required(VERSION 3.24)
project(AbsorbingEndpointFixture LANGUAGES C CXX)
option(ABSORB_CUDA "Compile the native endpoint kernel with CUDA" OFF)
if(ABSORB_CUDA)
    enable_language(CUDA)
endif()
find_package(AMReX CONFIG REQUIRED)
set(ABSORB_GEOMETRY RZ CACHE STRING "RZ or 3D")
get_filename_component(WARPX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(test_absorbing_endpoint test_absorbing_endpoint.cpp)
target_link_libraries(test_absorbing_endpoint PRIVATE AMReX::amrex)
target_compile_features(test_absorbing_endpoint PRIVATE cxx_std_20)
target_include_directories(test_absorbing_endpoint PRIVATE ${WARPX_SOURCE}/Source)
target_compile_definitions(test_absorbing_endpoint PRIVATE WARPX_DIM_${ABSORB_GEOMETRY})
if(ABSORB_GEOMETRY STREQUAL "RZ")
    target_compile_definitions(test_absorbing_endpoint PRIVATE WARPX_ZINDEX=1)
else()
    target_compile_definitions(test_absorbing_endpoint PRIVATE WARPX_ZINDEX=2)
endif()
if(ABSORB_CUDA)
    set_source_files_properties(test_absorbing_endpoint.cpp PROPERTIES LANGUAGE CUDA)
    target_compile_options(test_absorbing_endpoint PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-extended-lambda;--expt-relaxed-constexpr>)
    set_target_properties(test_absorbing_endpoint PROPERTIES CUDA_STANDARD 20)
endif()
enable_testing()
add_test(NAME deterministic_absorbing_endpoint COMMAND test_absorbing_endpoint)
set_tests_properties(deterministic_absorbing_endpoint PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1")
