cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXCircuitMMIncrementFixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(CIRCUIT_MM_DENSITY_EXECUTABLE AND CIRCUIT_MM_RETRY_EXECUTABLE)
    foreach(kind IN ITEMS density retry)
        add_executable(test_circuit_mm_${kind} IMPORTED GLOBAL)
    endforeach()
    set_target_properties(test_circuit_mm_density PROPERTIES IMPORTED_LOCATION "${CIRCUIT_MM_DENSITY_EXECUTABLE}")
    set_target_properties(test_circuit_mm_retry PROPERTIES IMPORTED_LOCATION "${CIRCUIT_MM_RETRY_EXECUTABLE}")
elseif(TARGET lib_rz)
    set(helper_dir ${WarpX_SOURCE_DIR}/Source/FieldSolver/ImplicitSolvers)
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare_retry_hook.py
        ${helper_dir}/ImplicitSolver.cpp ${CMAKE_CURRENT_BINARY_DIR}/RetryImplicitSolver.cpp
        COMMAND_ERROR_IS_FATAL ANY)
    add_executable(test_circuit_mm_density density_main.cpp)
    add_executable(test_circuit_mm_retry retry_main.cpp ${CMAKE_CURRENT_BINARY_DIR}/RetryImplicitSolver.cpp)
    foreach(kind IN ITEMS density retry)
        target_link_libraries(test_circuit_mm_${kind} PRIVATE lib_rz)
        target_include_directories(test_circuit_mm_${kind} PRIVATE ${WarpX_SOURCE_DIR}/Source ${helper_dir})
        if(WarpX_COMPUTE STREQUAL "CUDA")
            setup_target_for_cuda_compilation(test_circuit_mm_${kind})
        endif()
    endforeach()
else()
    message(FATAL_ERROR "Requires coherent lib_rz or both CIRCUIT_MM_*_EXECUTABLE paths")
endif()
if(NOT CIRCUIT_MM_PLUGIN_LIBRARY OR NOT CIRCUIT_MM_PLUGIN_CONFIG)
    message(FATAL_ERROR "The native identity/rate/clock provider and pinned one-port model are required")
endif()
set(common --input ${CMAKE_CURRENT_SOURCE_DIR}/inputs --plugin ${CIRCUIT_MM_PLUGIN_LIBRARY}
    --config ${CIRCUIT_MM_PLUGIN_CONFIG})
function(circuit_mm_test name kind)
    add_test(NAME circuit_mm_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run_and_check.py --executable $<TARGET_FILE:test_circuit_mm_${kind}>
        ${common} ${ARGN})
    set_tests_properties(circuit_mm_${name} PROPERTIES TIMEOUT 900 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
circuit_mm_test(density density --case density)
circuit_mm_test(nonfinite density --case nonfinite)
circuit_mm_test(stale density --case stale)
foreach(mode IN ITEMS coupled_jfnk decoupled_jfnk)
    circuit_mm_test(retry_${mode} retry --case retry --mode ${mode})
endforeach()
if(MPIEXEC_EXECUTABLE)
    circuit_mm_test(density_mpi2 density --case density --ranks 2 --mpi ${MPIEXEC_EXECUTABLE})
    foreach(mode IN ITEMS coupled_jfnk decoupled_jfnk)
        circuit_mm_test(retry_${mode}_mpi2 retry --case retry --mode ${mode} --ranks 2 --mpi ${MPIEXEC_EXECUTABLE})
        set_tests_properties(circuit_mm_retry_${mode}_mpi2 PROPERTIES PROCESSORS 2)
    endforeach()
endif()
