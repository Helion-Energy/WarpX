cmake_minimum_required(VERSION 3.24)
project(WarpXEndpointAuditFixture LANGUAGES C CXX)
find_package(AMReX CONFIG REQUIRED)
get_filename_component(WARPX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(AUDIT_GEOMETRY RZ CACHE STRING "RZ or 3D")
enable_testing()
add_executable(test_endpoint_audit test_endpoint_audit.cpp)
target_link_libraries(test_endpoint_audit PRIVATE AMReX::amrex)
target_compile_features(test_endpoint_audit PRIVATE cxx_std_20)
target_include_directories(test_endpoint_audit PRIVATE ${WARPX_SOURCE}/Source
    ${CMAKE_CURRENT_SOURCE_DIR}/../thermal_moments_fixture)
target_compile_definitions(test_endpoint_audit PRIVATE WARPX_DIM_${AUDIT_GEOMETRY})
if(AUDIT_GEOMETRY STREQUAL RZ)
    target_compile_definitions(test_endpoint_audit PRIVATE WARPX_ZINDEX=1)
else()
    target_compile_definitions(test_endpoint_audit PRIVATE WARPX_ZINDEX=2)
endif()
foreach(case IN ITEMS interior axis periodic absorbing thermal reflecting wall near_wall resolved_wall
    gather_reach current_reach endpoint_reach projection_reach nonfinite identity old_outside aba)
    foreach(box IN ITEMS 4 64)
        add_test(NAME endpoint_${case}_b${box} COMMAND test_endpoint_audit
            test.case=${case} test.max_grid_size=${box})
        set_tests_properties(endpoint_${case}_b${box} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1")
    endforeach()
endforeach()
