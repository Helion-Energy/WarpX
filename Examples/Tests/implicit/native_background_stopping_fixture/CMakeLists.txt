cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXNativeBackgroundStoppingFixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NATIVE_BACKGROUND_STOPPING_EXECUTABLE)
    add_executable(test_native_background_stopping IMPORTED GLOBAL)
    set_target_properties(test_native_background_stopping PROPERTIES IMPORTED_LOCATION "${NATIVE_BACKGROUND_STOPPING_EXECUTABLE}")
elseif(TARGET lib_rz)
    add_executable(test_native_background_stopping main.cpp)
    target_link_libraries(test_native_background_stopping PRIVATE lib_rz)
    target_include_directories(test_native_background_stopping PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(WarpX_COMPUTE STREQUAL "CUDA")
        setup_target_for_cuda_compilation(test_native_background_stopping)
    endif()
else()
    message(FATAL_ERROR "Requires lib_rz or NATIVE_BACKGROUND_STOPPING_EXECUTABLE")
endif()
# Raw-state rollback is exact on all backends. The current fixture also
# requires exact CPU recomputation of native atomic deposition/solve outputs;
# a separate GPU arithmetic certificate is required for runtime registration.
if(WarpX_COMPUTE STREQUAL "CUDA")
    return()
endif()
function(background_case name)
    add_test(NAME native_background_stopping_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run.py --executable $<TARGET_FILE:test_native_background_stopping>
        --case ${name} ${ARGN})
    set_tests_properties(native_background_stopping_${name} PROPERTIES TIMEOUT 300 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
background_case(periodic_coupled --steps 2)
background_case(pmc_decoupled --pmc --mode decoupled_jfnk --steps 2)
background_case(new_api_zero --zero)
foreach(fault IN ITEMS ampere magnetic wall stale alias)
    background_case(reject_${fault} --negative ${fault})
endforeach()
if(MPIEXEC_EXECUTABLE)
    background_case(mpi_seam --grid 16 --ranks 2 --mpi ${MPIEXEC_EXECUTABLE})
    background_case(mpi_empty --pmc --ranks 2 --mpi ${MPIEXEC_EXECUTABLE})
    set_tests_properties(native_background_stopping_mpi_seam native_background_stopping_mpi_empty PROPERTIES PROCESSORS 2)
endif()
