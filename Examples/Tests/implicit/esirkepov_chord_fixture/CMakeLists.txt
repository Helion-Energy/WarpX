# CPU-native regression of the completed implicit orbit and stable spline algebra.
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EsirkepovChordQualification LANGUAGES NONE)
    enable_testing()
endif()
if(TARGET lib_rz AND NOT WarpX_COMPUTE STREQUAL "CUDA"
        AND WarpX_PRECISION STREQUAL "DOUBLE"
        AND WarpX_PARTICLE_PRECISION STREQUAL "DOUBLE")
    add_executable(test_esirkepov_chord_rz test_chord_capture.cpp)
    target_link_libraries(test_esirkepov_chord_rz PRIVATE lib_rz)
    target_include_directories(test_esirkepov_chord_rz PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        add_executable(test_esirkepov_shape_difference test_shape_difference.cpp)
        target_link_libraries(test_esirkepov_shape_difference PRIVATE lib_rz quadmath)
        target_include_directories(test_esirkepov_shape_difference PRIVATE ${WarpX_SOURCE_DIR}/Source)
    endif()
elseif(DEFINED ESIRKEPOV_CHORD_BINARY)
    add_executable(test_esirkepov_chord_rz IMPORTED)
    set_target_properties(test_esirkepov_chord_rz PROPERTIES IMPORTED_LOCATION "${ESIRKEPOV_CHORD_BINARY}")
endif()
if(DEFINED ESIRKEPOV_SHAPE_BINARY AND NOT TARGET test_esirkepov_shape_difference)
    add_executable(test_esirkepov_shape_difference IMPORTED)
    set_target_properties(test_esirkepov_shape_difference PROPERTIES IMPORTED_LOCATION "${ESIRKEPOV_SHAPE_BINARY}")
endif()
foreach(test IN ITEMS test_esirkepov_chord_rz test_esirkepov_shape_difference)
    if(TARGET ${test})
        add_test(NAME ${test} COMMAND $<TARGET_FILE:${test}>)
        set_tests_properties(${test} PROPERTIES TIMEOUT 60 ENVIRONMENT "OMP_NUM_THREADS=1")
    endif()
endforeach()
if(TARGET test_esirkepov_chord_rz AND MPIEXEC_EXECUTABLE)
    add_test(NAME test_esirkepov_chord_rz_mpi2 COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2 ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_esirkepov_chord_rz> ${MPIEXEC_POSTFLAGS})
    set_tests_properties(test_esirkepov_chord_rz_mpi2 PROPERTIES PROCESSORS 2 TIMEOUT 60 ENVIRONMENT "OMP_NUM_THREADS=1")
endif()
