cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXNativeRetainedAcceptanceFixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NATIVE_RETAINED_ACCEPTANCE_EXECUTABLE)
    add_executable(test_native_retained_acceptance IMPORTED GLOBAL)
    set_target_properties(test_native_retained_acceptance PROPERTIES IMPORTED_LOCATION "${NATIVE_RETAINED_ACCEPTANCE_EXECUTABLE}")
elseif(TARGET lib_rz)
    add_executable(test_native_retained_acceptance main.cpp)
    target_link_libraries(test_native_retained_acceptance PRIVATE lib_rz)
    target_include_directories(test_native_retained_acceptance PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(WarpX_COMPUTE STREQUAL "CUDA")
        setup_target_for_cuda_compilation(test_native_retained_acceptance)
    endif()
else()
    message(FATAL_ERROR "Requires lib_rz or NATIVE_RETAINED_ACCEPTANCE_EXECUTABLE")
endif()
# In-process transaction byte restoration is backend independent. The fresh
# retry comparison additionally demands CPU deterministic summation; this
# fixture does not assert GPU byte-identical atomic recomputation.
if(WarpX_COMPUTE STREQUAL "CUDA")
    return()
endif()
function(retained_case name)
    add_test(NAME native_retained_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run.py --executable $<TARGET_FILE:test_native_retained_acceptance>
        --case ${name} ${ARGN})
    set_tests_properties(native_retained_${name} PROPERTIES TIMEOUT 300 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
retained_case(periodic_coupled)
retained_case(pmc_decoupled_warm --pmc --mode decoupled_jfnk --steps 2 --reject-step 1)
retained_case(periodic_decoupled_constraint --mode decoupled_jfnk --fault companion)
retained_case(pmc_coupled_constraint_warm --pmc --steps 2 --reject-step 1 --fault companion)
retained_case(preflight --preflight)
if(MPIEXEC_EXECUTABLE)
    retained_case(mpi_rejected_relocation --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --seam)
    retained_case(mpi_empty_collective_veto --pmc --mode decoupled_jfnk --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --veto-rank 1)
    set_tests_properties(native_retained_mpi_rejected_relocation native_retained_mpi_empty_collective_veto PROPERTIES PROCESSORS 2)
endif()

retained_case(diagnostic_first --diagnostics)
retained_case(diagnostic_warm --diagnostics --mode decoupled_jfnk --steps 2 --reject-step 1 --fault companion)
if(MPIEXEC_EXECUTABLE)
    retained_case(diagnostic_empty_veto --diagnostics --mode decoupled_jfnk --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --veto-rank 1)
    set_tests_properties(native_retained_diagnostic_empty_veto PROPERTIES PROCESSORS 2)
endif()
