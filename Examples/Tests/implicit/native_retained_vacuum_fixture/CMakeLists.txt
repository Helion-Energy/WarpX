cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(WarpXNativeRetainedVacuumFixture LANGUAGES CXX)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NATIVE_RETAINED_VACUUM_EXECUTABLE)
    add_executable(test_native_retained_vacuum IMPORTED GLOBAL)
    set_target_properties(test_native_retained_vacuum PROPERTIES IMPORTED_LOCATION "${NATIVE_RETAINED_VACUUM_EXECUTABLE}")
elseif(TARGET lib_rz)
    add_executable(test_native_retained_vacuum main.cpp)
    target_link_libraries(test_native_retained_vacuum PRIVATE lib_rz)
    target_include_directories(test_native_retained_vacuum PRIVATE ${WarpX_SOURCE_DIR}/Source)
    if(WarpX_COMPUTE STREQUAL "CUDA")
        setup_target_for_cuda_compilation(test_native_retained_vacuum)
    endif()
else()
    message(FATAL_ERROR "Requires lib_rz or NATIVE_RETAINED_VACUUM_EXECUTABLE")
endif()
# In-process byte restoration is portable. These clean/retry comparisons retain
# the CPU exact arithmetic gate; parent separately owns GPU runtime checks.
if(WarpX_COMPUTE STREQUAL "CUDA")
    return()
endif()
function(vacuum_case name)
    add_test(NAME native_retained_vacuum_${name} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/run.py --executable $<TARGET_FILE:test_native_retained_vacuum>
        --case ${name} ${ARGN})
    set_tests_properties(native_retained_vacuum_${name} PROPERTIES TIMEOUT 300 ENVIRONMENT "OMP_NUM_THREADS=1")
endfunction()
vacuum_case(coupled_mm0_first --fault before)
vacuum_case(decoupled_mm0_warm --mode decoupled_jfnk --reject-step 1)
vacuum_case(coupled_mm1_density --mm 1 --fault density)
vacuum_case(decoupled_mm1_nonfinite_warm --mode decoupled_jfnk --mm 1 --fault nonfinite --reject-step 1)
vacuum_case(preflight --preflight)
vacuum_case(guard_projection --negative projection_without_mm)
vacuum_case(guard_joule --negative joule)
vacuum_case(guard_driver --negative driver)
vacuum_case(guard_nonzero_static --negative nonzero)
vacuum_case(guard_segment --negative segment)
if(MPIEXEC_EXECUTABLE)
    vacuum_case(mpi_coupled_mm0_relocate --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --relocate)
    vacuum_case(mpi_decoupled_mm0_empty --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mode decoupled_jfnk --veto-rank 1 --reject-step 1)
    vacuum_case(mpi_coupled_mm1_density --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mm 1 --fault density)
    vacuum_case(mpi_decoupled_mm1_relocate_warm --ranks 2 --mpi ${MPIEXEC_EXECUTABLE} --mode decoupled_jfnk --mm 1 --relocate --reject-step 1)
    set_tests_properties(native_retained_vacuum_mpi_coupled_mm0_relocate native_retained_vacuum_mpi_decoupled_mm0_empty native_retained_vacuum_mpi_coupled_mm1_density native_retained_vacuum_mpi_decoupled_mm1_relocate_warm PROPERTIES PROCESSORS 2)
endif()
