cmake_minimum_required(VERSION 3.24)
project(IonElectricWorkFixture LANGUAGES C CXX)
find_package(AMReX CONFIG REQUIRED)
get_filename_component(WARPX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WORK_GEOMETRY RZ CACHE STRING "RZ or 3D")
add_executable(test_ion_electric_work test_ion_electric_work.cpp
 ${WARPX_SOURCE}/Source/FieldSolver/ImplicitSolvers/ImplicitIonElectricWork.cpp)
target_link_libraries(test_ion_electric_work PRIVATE AMReX::amrex)
target_compile_features(test_ion_electric_work PRIVATE cxx_std_20)
target_include_directories(test_ion_electric_work PRIVATE ${WARPX_SOURCE}/Source)
target_compile_definitions(test_ion_electric_work PRIVATE WARPX_DIM_${WORK_GEOMETRY})
if(WORK_GEOMETRY STREQUAL RZ)
 target_compile_definitions(test_ion_electric_work PRIVATE WARPX_ZINDEX=1)
else()
 target_compile_definitions(test_ion_electric_work PRIVATE WARPX_ZINDEX=2)
endif()
enable_testing()
foreach(case IN ITEMS recorded rotation cooling cancellation reconstruction invalid_field invalid_particle ghosts metadata no_total derivative matched_limit remainder partial_remainder)
 foreach(box IN ITEMS 4 64)
  add_test(NAME electric_work_${case}_box${box} COMMAND test_ion_electric_work test.case=${case} test.box=${box})
  set_tests_properties(electric_work_${case}_box${box} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1")
 endforeach()
endforeach()
