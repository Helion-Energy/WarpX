cmake_minimum_required(VERSION 3.24)
project(WarpXGatherSafetyFixture LANGUAGES C CXX)
find_package(AMReX CONFIG REQUIRED)
get_filename_component(WARPX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(GATHER_GEOMETRY RZ CACHE STRING "RZ or 3D")
option(GATHER_SANITIZE "Enable CPU address/undefined/float conversion sanitizers" OFF)
enable_testing()
add_executable(test_gather_safety test_gather_safety.cpp)
target_link_libraries(test_gather_safety PRIVATE AMReX::amrex)
target_compile_features(test_gather_safety PRIVATE cxx_std_20)
target_include_directories(test_gather_safety PRIVATE ${WARPX_SOURCE}/Source)
target_compile_definitions(test_gather_safety PRIVATE WARPX_DIM_${GATHER_GEOMETRY})
if(GATHER_GEOMETRY STREQUAL RZ)
    target_compile_definitions(test_gather_safety PRIVATE WARPX_ZINDEX=1)
else()
    target_compile_definitions(test_gather_safety PRIVATE WARPX_ZINDEX=2)
endif()
if(GATHER_SANITIZE)
    target_compile_options(test_gather_safety PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
    target_link_options(test_gather_safety PRIVATE -fsanitize=address,undefined,float-cast-overflow)
endif()
foreach(order RANGE 1 4)
    foreach(layout IN ITEMS nodal yee)
        foreach(box IN ITEMS 4 64)
            add_test(NAME gather_${layout}_order${order}_box${box}
                COMMAND test_gather_safety test.order=${order} test.layout=${layout} test.box=${box})
            set_tests_properties(gather_${layout}_order${order}_box${box} PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=1")
        endforeach()
    endforeach()
endforeach()
